export type SourceIdentity = { path: string; sha256: string };
type Wire = ReturnType<typeof wire>;
type WireTriple = readonly [Wire, Wire, Wire];
export type PhysicalEvidence = {
  version: string;
  generatedAt: string;
  computeMs: number;
  sources: SourceIdentity[];
  sourceHash: string;
  cohorts: readonly [WireTriple, WireTriple, WireTriple];
  rate: WireTriple;
};
export type EvidenceProvenance = {
  version: string;
  generation: {
    sources: SourceIdentity[];
    sourceHash: string;
    isolatedCacheEmpty: true;
    physicalEnumeration: "full-ordered";
    oracleRows: 24;
    independentWaiting: "UNVERIFIED_NOT_REGENERATED";
  };
  historical: {
    sources: typeof ARCHIVED_SOURCES;
    files: typeof HISTORICAL_FILES;
  };
  physicalEnumeration: SourceIdentity & {
    cachePath: string;
    sourceHash: string;
    sources: SourceIdentity[];
  };
  panel: SourceIdentity;
};
export const sha256 = (bytes: Buffer | string) => createHash("sha256").update(bytes).digest("hex");
export function sourceIdentity(paths: readonly string[], root = process.cwd()) {
  const sources = paths.map((path) => ({
    path,
    sha256: sha256(readFileSync(resolve(root, path))),
  }));
  return { sources, sourceHash: sha256(JSON.stringify(sources)) };
}
export function pinnedBytes(identity: SourceIdentity, root = process.cwd()) {
  const bytes = readFileSync(resolve(root, identity.path));
  assert.equal(sha256(bytes), identity.sha256, `Independent physical hash drift: ${identity.path}`);
  return bytes;
}
export function authenticateHistoricalArchive(root = process.cwd()) {
  for (const entry of HISTORICAL_FILES) pinnedBytes(entry, root);
  for (const entry of ARCHIVED_SOURCES)
    pinnedBytes({ path: entry.archivePath, sha256: entry.sha256 }, root);
  const physical = JSON.parse(
    readFileSync(resolve(root, `${HISTORICAL_ROOT}/independent-physical-cohorts.json`), "utf8"),
  ) as PhysicalEvidence;
  const sources = ARCHIVED_SOURCES.map(({ path, sha256 }) => ({ path, sha256 }));
  assert.equal(physical.version, "independent-ordered-physical-supply-rates-v1");
  assert.deepEqual(physical.sources, sources, "Historical physical source identity drift");
  assert.equal(physical.sourceHash, sha256(JSON.stringify(sources)));
  return physical;
}
/** Compare exact values by reduced rational identity, never JSON representation. */
export function normalizeExact(value: unknown): unknown {
  if (Array.isArray(value)) return value.map(normalizeExact);
  if (value !== null && typeof value === "object") {
    if ("numerator" in value && "denominator" in value) {
      assert.equal(typeof value.numerator, "string");
      assert.equal(typeof value.denominator, "string");
      assert.deepEqual(Object.keys(value).sort(), ["denominator", "numerator"]);
      return wire(fromWire(value as Wire));
    }
    return Object.fromEntries(
      Object.entries(value).map(([key, item]) => [key, normalizeExact(item)]),
    );
  }
  return value;
}
export function validatePhysicalEvidence(physical: PhysicalEvidence, root = process.cwd()) {
  const identity = sourceIdentity(PHYSICAL_SOURCES, root);
  assert.equal(physical.version, PHYSICAL_VERSION, "Independent physical source identity drift");
  assert.deepEqual(
    physical.sources,
    identity.sources,
    "Independent physical source identity drift",
  );
  assert.equal(
    physical.sourceHash,
    identity.sourceHash,
    "Independent physical source identity drift",
  );
  assert.equal(physical.cohorts.length, 3, "Independent physical exact rate identity drift");
  for (const triple of [physical.rate, ...physical.cohorts]) {
    assert.equal(triple.length, 3, "Independent physical exact rate identity drift");
    for (const value of triple) {
      const exact = fromWire(value);
      assert.ok(exact.n >= 0n && exact.d > 0n, "Independent physical exact rate identity drift");
      assert.deepEqual(wire(exact), value, "Independent physical exact rate identity drift");
    }
  }
  assert.ok(Number.isFinite(physical.computeMs) && physical.computeMs > 0);
  assert.ok(Number.isFinite(Date.parse(physical.generatedAt)));
  return identity;
}
export function readV2Evidence(root = process.cwd()) {
  const pins = JSON.parse(readFileSync(resolve(root, `${V2_ROOT}/pins.json`), "utf8")) as {
    version: string;
    provenance: SourceIdentity;
  };
  assert.equal(pins.version, EVIDENCE_VERSION, "Independent physical pin identity drift");
  assert.equal(pins.provenance.path, `${V2_ROOT}/provenance.json`);
  const provenance = JSON.parse(
    pinnedBytes(pins.provenance, root).toString(),
  ) as EvidenceProvenance;
  assert.equal(provenance.version, EVIDENCE_VERSION);
  assert.deepEqual(provenance.historical, { sources: ARCHIVED_SOURCES, files: HISTORICAL_FILES });
  const historical = authenticateHistoricalArchive(root);
  const generation = sourceIdentity(GENERATION_SOURCES, root);
  assert.deepEqual(
    provenance.generation,
    {
      ...generation,
      isolatedCacheEmpty: true,
      physicalEnumeration: "full-ordered",
      oracleRows: 24,
      independentWaiting: "UNVERIFIED_NOT_REGENERATED",
    },
    "Independent physical generation source identity drift",
  );
  assert.equal(provenance.physicalEnumeration.path, `${V2_ROOT}/independent-physical-cohorts.json`);
  assert.equal(provenance.physicalEnumeration.cachePath, PHYSICAL_CACHE);
  const physicalBytes = pinnedBytes(provenance.physicalEnumeration, root);
  const physical = JSON.parse(physicalBytes.toString()) as PhysicalEvidence;
  const identity = validatePhysicalEvidence(physical, root);
  assert.deepEqual(provenance.physicalEnumeration.sources, identity.sources);
  assert.equal(provenance.physicalEnumeration.sourceHash, identity.sourceHash);
  assert.deepEqual(normalizeExact(physical.cohorts), normalizeExact(historical.cohorts));
  assert.deepEqual(normalizeExact(physical.rate), normalizeExact(historical.rate));
  assert.equal(provenance.panel.path, `${V2_ROOT}/panel.json`);
  const panel = JSON.parse(pinnedBytes(provenance.panel, root).toString()) as unknown[];
  assert.equal(panel.length, 24);
  assert.deepEqual(
    normalizeExact(panel),
    normalizeExact(
      JSON.parse(readFileSync(resolve(root, `${HISTORICAL_ROOT}/panel.json`), "utf8")),
    ),
    "Approved24 historical exact equivalence drift",
  );
  return { physical, physicalBytes, panel, provenance };
}
