export type SourceIdentity = { path: string; sha256: string };
export const sha256 = (bytes: Buffer | string) => createHash("sha256").update(bytes).digest("hex");
export function pinnedBytes(identity: SourceIdentity, root = process.cwd()) {
  const bytes = readFileSync(resolve(root, identity.path));
  assert.equal(sha256(bytes), identity.sha256, `Independent physical hash drift: ${identity.path}`);
  return bytes;
}
export function sourceIdentity(paths: readonly string[], root = process.cwd()) {
  const sources = paths.map((path) => ({
    path,
    sha256: sha256(readFileSync(resolve(root, path))),
  }));
  return { sources, sourceHash: sha256(JSON.stringify(sources)) };
}

type HistoricalPhysical = {
  version: string;
  sources: SourceIdentity[];
  sourceHash: string;
  cohorts: unknown;
  rate: unknown;
};
export function readHistorical(root = process.cwd()) {
  const files = new Map<string, Buffer>(
    HISTORICAL_FILES.map((entry) => [entry.path, pinnedBytes(entry, root)]),
  );
  for (const entry of ARCHIVED_SOURCES)
    pinnedBytes({ path: entry.archivePath, sha256: entry.sha256 }, root);
  const bytes = (path: string) => {
    const value = files.get(path);
    assert.ok(value, `Missing authenticated historical input: ${path}`);
    return value;
  };
  const physical = JSON.parse(
    bytes(`${HISTORICAL_ROOT}/independent-physical-cohorts.json`).toString(),
  ) as HistoricalPhysical;
  // The original seal recorded two sources. The recovered game dependency is
  // authenticated separately; it is not retroactively claimed as an original pin.
  const sources = ARCHIVED_SOURCES.slice(0, 2).map(({ path, sha256 }) => ({ path, sha256 }));
  assert.equal(physical.version, "independent-ordered-physical-supply-rates-v1");
  assert.deepEqual(physical.sources, sources);
  assert.equal(physical.sourceHash, sha256(JSON.stringify(sources)));
  return {
    physical,
    panel: JSON.parse(bytes(`${HISTORICAL_ROOT}/panel.json`).toString()) as unknown[],
    snapshotBytes: bytes(`${HISTORICAL_ROOT}/snapshot.json`),
  };
}

function readArchivedPublication(
  directory: string,
  pinsSha256: string,
  version: string,
  sourcePaths: readonly string[],
  root: string,
) {
  const pinsBytes = pinnedBytes({ path: `${directory}/pins.json.txt`, sha256: pinsSha256 }, root);
  const pins = JSON.parse(pinsBytes.toString()) as { version: string; provenance: SourceIdentity };
  assert.equal(pins.version, version);
  assert.equal(pins.provenance.path, `${directory}/provenance.json`);
  const provenance = JSON.parse(
    pinnedBytes({ ...pins.provenance, path: `${pins.provenance.path}.txt` }, root).toString(),
  ) as {
    generation: { sources: SourceIdentity[]; sourceHash: string; independentWaiting: string };
    physicalEnumeration: SourceIdentity;
    panel: SourceIdentity;
  };
  assert.deepEqual(
    provenance.generation.sources.map(({ path }) => path),
    sourcePaths,
  );
  assert.equal(
    provenance.generation.sourceHash,
    sha256(JSON.stringify(provenance.generation.sources)),
  );
  for (const source of provenance.generation.sources)
    pinnedBytes({ ...source, path: `${directory}/source/${source.path}.txt` }, root);
  assert.equal(
    provenance.physicalEnumeration.path,
    `${directory}/independent-physical-cohorts.json`,
  );
  assert.equal(provenance.panel.path, `${directory}/panel.json`);
  assert.equal(provenance.generation.independentWaiting, "UNVERIFIED_NOT_REGENERATED");
  const physicalBytes = pinnedBytes(
    { ...provenance.physicalEnumeration, path: `${provenance.physicalEnumeration.path}.txt` },
    root,
  );
  return {
    provenance,
    physical: JSON.parse(physicalBytes.toString()) as HistoricalPhysical,
    physicalBytes,
    panel: JSON.parse(
      pinnedBytes({ ...provenance.panel, path: `${provenance.panel.path}.txt` }, root).toString(),
    ) as unknown[],
  };
}
